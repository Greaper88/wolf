cmake_minimum_required(VERSION 3.16)
project(wolf_gpu_policy_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
enable_testing()
set(GPU ../../src/moonlight-server/gpu)
add_executable(gpu_policy_tests selector_test.cpp ${GPU}/selector.cpp ${GPU}/discovery.cpp ${GPU}/admission.cpp)
target_include_directories(gpu_policy_tests PRIVATE ${GPU}/..)
target_link_libraries(gpu_policy_tests PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(gpu_policy_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME gpu_policy_tests COMMAND gpu_policy_tests)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_library(fake_nvml SHARED fake_nvml.cpp)
  set_target_properties(fake_nvml PROPERTIES OUTPUT_NAME nvidia-ml SOVERSION 1)
  set_tests_properties(gpu_policy_tests PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=$<TARGET_FILE_DIR:fake_nvml>;WOLF_GPU_TEST_NVML=1")
endif()

# Optional real GStreamer API tests; the policy suite remains dependency-free.
option(TEST_GPU_ENCODER_PROBE "Build GStreamer device-binding/probe tests" OFF)
if(TEST_GPU_ENCODER_PROBE AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GPU_GST REQUIRED IMPORTED_TARGET gstreamer-1.0)
  add_executable(gpu_encoder_tests encoder_probe_test.cpp ${GPU}/encoder_probe.cpp ${GPU}/selector.cpp)
  pkg_get_variable(GPU_TEST_PLUGIN_DIR gstreamer-1.0 pluginsdir)
  target_compile_definitions(gpu_encoder_tests PRIVATE GPU_TEST_PLUGIN_DIR="${GPU_TEST_PLUGIN_DIR}")
  target_include_directories(gpu_encoder_tests PRIVATE ${GPU}/..)
  target_link_libraries(gpu_encoder_tests PRIVATE PkgConfig::GPU_GST ${CMAKE_DL_LIBS})
  target_compile_options(gpu_encoder_tests PRIVATE -Wall -Wextra -Wpedantic)
  add_library(fake_cuda SHARED fake_cuda.cpp)
  set_target_properties(fake_cuda PROPERTIES OUTPUT_NAME cuda SOVERSION 1)
  add_test(NAME gpu_encoder_tests COMMAND gpu_encoder_tests)
  set_tests_properties(gpu_encoder_tests PROPERTIES ENVIRONMENT
    "LD_LIBRARY_PATH=$<TARGET_FILE_DIR:fake_cuda>;GST_PLUGIN_SYSTEM_PATH=;GST_PLUGIN_PATH=;GST_REGISTRY=${CMAKE_CURRENT_BINARY_DIR}/test-registry.bin")
endif()
if(TEST_GPU_ENCODER_PROBE AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(wolf-gpu-probe ../../tools/gpu-probe.cpp ${GPU}/encoder_probe.cpp ${GPU}/probe_child.cpp ${GPU}/probe_process.cpp ${GPU}/selector.cpp ${GPU}/discovery.cpp)
  target_include_directories(wolf-gpu-probe PRIVATE ${GPU}/..)
  target_link_libraries(wolf-gpu-probe PRIVATE PkgConfig::GPU_GST ${CMAKE_DL_LIBS})
  target_compile_options(wolf-gpu-probe PRIVATE -Wall -Wextra -Wpedantic)
endif()
